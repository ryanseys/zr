# The program tests that can fail on wasm32-wasi, and why. Each line is
# a program's path below test/programs and a reason, separated by a tab.
# A `#` line is a comment (test/support/baseline.rb).
#
# The run skips a listed test that fails, and prints its reason. A
# listed test that PASSES fails the run, with a message to remove it
# from this file. Thus the file cannot become out of date.

# wasi has no processes and no signals.
cli/core/process/process_spawn.rb	spawn() is unimplemented on wasi
cli/core/process/signal_trap.rb	kill has no wasm form
cli/core/kernel/recursion_depth_follows_the_vm_stack_size.rb	IO.popen: pipe has no wasm form
cli/lang/require/a_cleared_load_path_finds_no_file_of_rubys_library.rb	IO.popen: pipe has no wasm form
cli/lang/require/with_gems_off_rubygems_defines_gem_over_nothing.rb	IO.popen: pipe has no wasm form
cli/stdlib/rbconfig/ruby_is_bindir_and_ruby_install_name.rb	RbConfig.ruby: a wasm guest has no ruby executable

# wasi cannot allocate a fiber stack, so no Thread or Fiber can start.
cli/core/fiber/fiber_resume.rb	can't allocate fiber stack: Function not implemented
cli/core/thread/threads_join.rb	can't allocate fiber stack: Function not implemented
cli/core/thread/thread_queue_handoff.rb	can't allocate fiber stack: Function not implemented
lang/methods/super_in_a_block_that_another_thread_runs_goes_on_from_the_def_s_node.rb	can't allocate fiber stack: Function not implemented
lang/methods/a_def_in_a_define_method_body_defines_where_the_block_was_written.rb	syntax_suggest runs in Timeout.timeout, which needs a Thread, so the report is prism's alone

# On wasi, eval uses the tree-walking interpreter (lib/zr/eval/interpreter.rb).
# These programs use eval, but eval support does not cause their
# failure.
cli/stdlib/ffi/ffi_libc_call.rb	dlopen has no wasm form

# A method that an eval defines runs in the walk's frames: an error in it
# names the walk's methods (Tree#bind), where ruby names the def. Its
# parameters, arity and inspect are the def's (World.describe).
lang/methods/a_block_parameter_of_nil_refuses_a_block_after_the_arity_and_keyword_checks.rb	the backtrace label of an eval'd def is the tree walk's
