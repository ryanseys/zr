# The end-to-end programs wasm32-wasi is allowed to fail, and why. One
# path and reason a line, separated by a tab; a `#` line is a comment.
#
# A named program that fails is skipped with its reason. A named program
# that PASSES fails the run and says to take it off this file, so a row
# cannot go stale without anyone noticing.

# wasi has no processes and no signals.
e2e/core/process/process_spawn.rb	spawn() is unimplemented on wasi
e2e/core/process/signal_trap.rb	kill has no wasm form

# wasi has no way to allocate a fiber stack, so every Thread and Fiber stops.
e2e/core/fiber/fiber_resume.rb	can't allocate fiber stack: Function not implemented
e2e/core/thread/threads_join.rb	can't allocate fiber stack: Function not implemented

# The wasm road has no unit 120 of the runtime, so anything that reaches
# it fails to link.
e2e/core/io/file_io_roundtrip.rb	the build fails: no data named zr_road_unit_120
e2e/lang/eval/eval_in_a_binary.rb	the build fails: no data named zr_road_unit_120
e2e/lang/require/require_a_bundled_library.rb	the build fails: no data named zr_road_unit_120
e2e/stdlib/ffi/ffi_libc_call.rb	the build fails: no data named zr_road_unit_120
e2e/stdlib/rubygems/require_rubygems.rb	the build fails: no data named zr_road_unit_120

# A 32-bit module wants more memory than the bound allows, to build or to run.
e2e/stdlib/bundler/bundler_resolves_a_gemfile.rb	the build holds more than 2048 MB
e2e/stdlib/bundler/require_bundler.rb	the build holds more than 3072 MB
e2e/stdlib/testing/a_minitest_suite_runs.rb	the build holds more than 2048 MB
e2e/core/thread/thread_queue_handoff.rb	the program holds more than 1024 MB
e2e/stdlib/zlib/zlib_roundtrip.rb	the program holds more than 1024 MB

# socket's wasi half is not written: `socket/constants` has no wasi table.
e2e/stdlib/socket/socket_loopback.rb	cannot load such file -- socket/constants
